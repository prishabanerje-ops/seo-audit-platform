// backend/services/ads.crypto.js
import { createCipheriv, createDecipheriv, randomBytes } from 'crypto';

/**
 * Derive a 32-byte key from ADS_TOKEN_ENCRYPTION_KEY (preferred)
 * or fall back to the first 32 bytes of JWT_SECRET.
 */
function getKey() {
  const raw = process.env.ADS_TOKEN_ENCRYPTION_KEY || process.env.JWT_SECRET || '';
  const hex = raw.replace(/[^0-9a-fA-F]/g, '').slice(0, 64);
  if (hex.length < 64) throw new Error('ADS_TOKEN_ENCRYPTION_KEY or JWT_SECRET must be at least 64 hex chars');
  return Buffer.from(hex, 'hex'); // 32 bytes
}

/**
 * Encrypt a plaintext string. Returns base64: iv:authTag:ciphertext
 */
export function encryptToken(plaintext) {
  const key = getKey();
  const iv = randomBytes(12); // 96-bit IV for GCM
  const cipher = createCipheriv('aes-256-gcm', key, iv);
  const encrypted = Buffer.concat([cipher.update(plaintext, 'utf8'), cipher.final()]);
  const authTag = cipher.getAuthTag();
  return [iv, authTag, encrypted].map(b => b.toString('base64')).join(':');
}

/**
 * Decrypt a token encrypted by encryptToken.
 */
export function decryptToken(combined) {
  const [ivB64, tagB64, dataB64] = combined.split(':');
  const key = getKey();
  const iv = Buffer.from(ivB64, 'base64');
  const authTag = Buffer.from(tagB64, 'base64');
  const data = Buffer.from(dataB64, 'base64');
  const decipher = createDecipheriv('aes-256-gcm', key, iv);
  decipher.setAuthTag(authTag);
  return Buffer.concat([decipher.update(data), decipher.final()]).toString('utf8');
}
```

// backend/services/ads.service.js
import { randomBytes } from 'crypto';
import db from '../db/sqlite.js';
import { encryptToken, decryptToken } from './ads.crypto.js';

const IST_OFFSET_MINUTES = 330; // UTC+5:30

/**
 * Returns { startDate, endDate } as 'YYYY-MM-DD' strings in IST.
 * @param {'last7d'|'last30d'|'thisMonth'} range
 */
export function getDateRange(range) {
  const nowUtc = new Date();
  // Compute "today" in IST
  const istNow = new Date(nowUtc.getTime() + IST_OFFSET_MINUTES * 60 * 1000);
  const year  = istNow.getUTCFullYear();
  const month = istNow.getUTCMonth(); // 0-based
  const day   = istNow.getUTCDate();

  const todayIST = new Date(Date.UTC(year, month, day));

  let startDate;
  const endDate = todayIST;

  if (range === 'last7d') {
    startDate = new Date(Date.UTC(year, month, day - 6));
  } else if (range === 'last30d') {
    startDate = new Date(Date.UTC(year, month, day - 29));
  } else if (range === 'thisMonth') {
    startDate = new Date(Date.UTC(year, month, 1));
  } else {
    throw new Error(`Unknown date range: ${range}`);
  }

  const fmt = d => d.toISOString().slice(0, 10);
  return { startDate: fmt(startDate), endDate: fmt(endDate) };
}

// ── OAuth State ────────────────────────────────────────────────────────────

/**
 * Generate and store a 32-byte hex state token for OAuth CSRF protection.
 * TTL: 10 minutes.
 */
export function createOAuthState(userId, platform) {
  const state = randomBytes(32).toString('hex');
  const expiresAt = new Date(Date.now() + 10 * 60 * 1000).toISOString();
  db.prepare('INSERT INTO ad_oauth_state (state, user_id, platform, expires_at) VALUES (?, ?, ?, ?)')
    .run(state, userId, platform, expiresAt);
  return state;
}

/**
 * Validate and consume a state token. Throws if invalid or expired.
 * Returns { userId, platform }.
 */
export function consumeOAuthState(state) {
  const row = db.prepare('SELECT * FROM ad_oauth_state WHERE state = ?').get(state);
  if (!row) throw new Error('Invalid OAuth state');
  db.prepare('DELETE FROM ad_oauth_state WHERE state = ?').run(state);
  if (new Date(row.expires_at) < new Date()) throw new Error('OAuth state expired');
  return { userId: row.user_id, platform: row.platform };
}

// ── Connections ───────────────────────────────────────────────────────────

/**
 * Save or update an OAuth connection for a company+platform.
 * Tokens are encrypted at rest.
 */
export function saveConnection({ companyId, platform, accessToken, refreshToken, tokenExpiresAt }) {
  const id = randomBytes(16).toString('hex');
  const encAccess  = encryptToken(accessToken);
  const encRefresh = refreshToken ? encryptToken(refreshToken) : null;

  db.prepare(`
    INSERT INTO ad_connections (id, company_id, platform, access_token, refresh_token, token_expires_at)
    VALUES (?, ?, ?, ?, ?, ?)
    ON CONFLICT(company_id, platform) DO UPDATE SET
      access_token     = excluded.access_token,
      refresh_token    = excluded.refresh_token,
      token_expires_at = excluded.token_expires_at,
      updated_at       = datetime('now')
  `).run(id, companyId, platform, encAccess, encRefresh, tokenExpiresAt || null);

  return db.prepare('SELECT * FROM ad_connections WHERE company_id = ? AND platform = ?')
    .get(companyId, platform);
}

/**
 * Get a decrypted connection. Returns null if not connected.
 */
export function getConnection(companyId, platform) {
  const row = db.prepare('SELECT * FROM ad_connections WHERE company_id = ? AND platform = ?')
    .get(companyId, platform);
  if (!row) return null;
  return {
    ...row,
    access_token:  decryptToken(row.access_token),
    refresh_token: row.refresh_token ? decryptToken(row.refresh_token) : null,
  };
}

/**
 * List all connections for a company (tokens NOT decrypted — for display only).
 */
export function listConnections(companyId) {
  return db.prepare('SELECT id, company_id, platform, selected_account_id, created_at, updated_at FROM ad_connections WHERE company_id = ?')
    .all(companyId);
}

/**
 * Delete a connection and cascade to accounts/snapshots.
 */
export function deleteConnection(companyId, platform) {
  db.prepare('DELETE FROM ad_connections WHERE company_id = ? AND platform = ?')
    .run(companyId, platform);
}

// ── Snapshot Cache ─────────────────────────────────────────────────────────

/**
 * Store metrics JSON for a connection + date range (upsert).
 */
export function saveSnapshot(connectionId, dateRange, data) {
  const id = randomBytes(16).toString('hex');
  db.prepare(`
    INSERT INTO ad_snapshots (id, connection_id, date_range, data, fetched_at)
    VALUES (?, ?, ?, ?, datetime('now'))
    ON CONFLICT(connection_id, date_range) DO UPDATE SET
      data       = excluded.data,
      fetched_at = excluded.fetched_at
  `).run(id, connectionId, dateRange, JSON.stringify(data));
}

/**
 * Get a cached snapshot if it's less than 1 hour old.
 * Returns parsed data or null.
 */
export function getCachedSnapshot(connectionId, dateRange) {
  const row = db.prepare(`
    SELECT data, fetched_at FROM ad_snapshots
    WHERE connection_id = ? AND date_range = ?
      AND fetched_at > datetime('now', '-1 hour')
  `).get(connectionId, dateRange);
  return row ? JSON.parse(row.data) : null;
}

const GOOGLE_STATUS_MAP = {
  ENABLED: 'active',
  PAUSED:  'paused',
  REMOVED: 'ended',

/**
 * Fetch metrics for the selected Google Ads account.
 * Returns normalized platform data object.
 *
 * @param {object} connection — decrypted connection from getConnection()
 * @param {string} startDate — YYYY-MM-DD
 * @param {string} endDate   — YYYY-MM-DD
 */
export async function fetchGoogleMetrics(connection, startDate, endDate) {
  if (!connection.selected_account_id) {
    throw new Error('No Google Ads account selected');
  }

  const client = new GoogleAdsApi({
    client_id:     process.env.GOOGLE_CLIENT_ID,
    client_secret: process.env.GOOGLE_CLIENT_SECRET,
    developer_token: process.env.GOOGLE_ADS_DEVELOPER_TOKEN,
  });

  const customer = client.Customer({
    customer_id:   connection.selected_account_id,
    refresh_token: connection.refresh_token,
  });

  const rows = await customer.query(`
    SELECT
      campaign.id,
      campaign.name,
      campaign.status,
      ad_group.id,
      ad_group.name,
      ad_group.status,
      ad_group_ad.ad.id,
      ad_group_ad.ad.name,
      ad_group_ad.status,
      metrics.clicks,
      metrics.impressions,
      metrics.conversions,
      metrics.cost_micros
    FROM ad_group_ad
    WHERE segments.date BETWEEN '${startDate}' AND '${endDate}'
  `);

  // Build nested structure: campaigns → adSets → ads
  const campaignMap = new Map();

  for (const row of rows) {
    const cId = String(row.campaign.id);
    if (!campaignMap.has(cId)) {
      campaignMap.set(cId, {
        id:     cId,
        name:   row.campaign.name,
        status: GOOGLE_STATUS_MAP[row.campaign.status] || 'unknown',
        adSets: new Map(),
      });
    }
    const campaign = campaignMap.get(cId);

    const agId = String(row.ad_group.id);
    if (!campaign.adSets.has(agId)) {
      campaign.adSets.set(agId, {
        id:     agId,
        name:   row.ad_group.name,
        status: GOOGLE_STATUS_MAP[row.ad_group.status] || 'unknown',
        ads:    [],
      });
    }
    const adSet = campaign.adSets.get(agId);

    adSet.ads.push({
      id:          String(row.ad_group_ad.ad.id),
      name:        row.ad_group_ad.ad.name || `Ad ${row.ad_group_ad.ad.id}`,
      status:      GOOGLE_STATUS_MAP[row.ad_group_ad.status] || 'unknown',
      clicks:      Number(row.metrics.clicks),
      impressions: Number(row.metrics.impressions),
      conversions: Number(row.metrics.conversions),
      spent:       Number(row.metrics.cost_micros) / 1_000_000,
    });
  }

  // Serialize Maps to arrays
  const campaigns = Array.from(campaignMap.values()).map(c => ({
    ...c,
    adSets: Array.from(c.adSets.values()),
  }));

  // Get account currency from ad_accounts table (already stored during connect)
  // Falls back to 'USD' if not found
  const db = (await import('../db/sqlite.js')).default;
  const account = db.prepare('SELECT currency, account_name FROM ad_accounts WHERE connection_id = ? AND is_selected = 1')
    .get(connection.id);

  return {
    platform:    'google',
    accountName: account?.account_name || 'Google Ads',
    currency:    account?.currency || 'USD',
    campaigns,
  };
}

const LI_BASE = 'https://api.linkedin.com/v2';
const LI_VERSION = '202501';

const LI_STATUS_MAP = {
  ACTIVE:   'active',
  PAUSED:   'paused',
  ARCHIVED: 'ended',
  CANCELED: 'ended',
  DRAFT:    'paused',
  COMPLETED:'ended',
};

function liHeaders(accessToken) {
  return {
    Authorization:      `Bearer ${accessToken}`,
    'LinkedIn-Version': LI_VERSION,
    'X-Restli-Protocol-Version': '2.0.0',
  };
}

/**
 * Fetch metrics from LinkedIn Campaign Manager for the selected account.
 */
export async function fetchLinkedInMetrics(connection, startDate, endDate) {
  if (!connection.selected_account_id) {
    throw new Error('No LinkedIn ad account selected');
  }

  const token     = connection.access_token;
  const accountId = connection.selected_account_id;
  const headers   = liHeaders(token);

  // 1. Fetch campaign groups (map to campaigns)
  const groupsRes = await fetch(
    `${LI_BASE}/adCampaignGroups?q=search&search.account.values[0]=urn:li:sponsoredAccount:${accountId}`,
    { headers }
  );
  const groupsJson = await groupsRes.json();
  const groups = groupsJson.elements || [];

  // 2. Fetch campaigns (map to adSets)
  const campaignsRes = await fetch(
    `${LI_BASE}/adCampaigns?q=search&search.account.values[0]=urn:li:sponsoredAccount:${accountId}`,
    { headers }
  );
  const campaignsJson = await campaignsRes.json();
  const liCampaigns = campaignsJson.elements || [];

  // 3. Fetch creatives (map to ads)
  const creativesRes = await fetch(
    `${LI_BASE}/adCreatives?q=search&search.account.values[0]=urn:li:sponsoredAccount:${accountId}`,
    { headers }
  );
  const creativesJson = await creativesRes.json();
  const creatives = creativesJson.elements || [];

  // 4. Fetch analytics for creatives
  const creativeIds = creatives.map(c => c.id).join(',');
  const analyticsRes = await fetch(
    `${LI_BASE}/adAnalytics?q=analytics&pivot=CREATIVE&dateRange.start.year=${startDate.slice(0,4)}&dateRange.start.month=${parseInt(startDate.slice(5,7))}&dateRange.start.day=${parseInt(startDate.slice(8,10))}&dateRange.end.year=${endDate.slice(0,4)}&dateRange.end.month=${parseInt(endDate.slice(5,7))}&dateRange.end.day=${parseInt(endDate.slice(8,10))}&timeGranularity=ALL&creatives=List(${creativeIds})`,
    { headers }
  );
  const analyticsJson = await analyticsRes.json();
  const analyticsElements = analyticsJson.elements || [];

  // Map creative URN → analytics
  const analyticsMap = new Map(
    analyticsElements.map(a => [a.pivotValues?.[0], a])
  );

  // Map campaign → group
  const campaignGroupMap = new Map(
    liCampaigns.map(c => [c.id, c.campaignGroup])
  );

  // Map creative → campaign
  const creativeCampaignMap = new Map(
    creatives.map(c => [c.id, c.campaign])
  );

  // Build nested structure: groups → campaigns → creatives
  const groupMap = new Map(
    groups.map(g => ({
      id:     g.id,
      name:   g.name,
      status: LI_STATUS_MAP[g.status] || 'unknown',
      adSets: new Map(),
    })).map(g => [g.id, g])
  );

  for (const campaign of liCampaigns) {
    const groupUrn = campaignGroupMap.get(campaign.id);
    const groupId = groupUrn?.split(':').pop();
    if (!groupMap.has(groupId)) continue;
    const group = groupMap.get(groupId);
    group.adSets.set(campaign.id, {
      id:     String(campaign.id),
      name:   campaign.name,
      status: LI_STATUS_MAP[campaign.status] || 'unknown',
      ads:    [],
    });
  }

  for (const creative of creatives) {
    const campaignUrn = creativeCampaignMap.get(creative.id);
    const campaignId = campaignUrn?.split(':').pop();
    const groupUrn = campaignGroupMap.get(campaignId);
    const groupId = groupUrn?.split(':').pop();
    if (!groupMap.has(groupId)) continue;
    const group = groupMap.get(groupId);
    if (!group.adSets.has(campaignId)) continue;
    const adSet = group.adSets.get(campaignId);

    const analytics = analyticsMap.get(creative.id) || {};
    adSet.ads.push({
      id:          String(creative.id),
      name:        creative.name || `Creative ${creative.id}`,
      status:      LI_STATUS_MAP[creative.status] || 'unknown',
      clicks:      Number(analytics.clicks || 0),
      impressions: Number(analytics.impressions || 0),
      conversions: Number(analytics.externalWebsiteConversions || 0),

/**
 * Force-refresh metrics for all connected platforms, bypassing cache.
 * Emits Socket.io events to the company room as each platform completes.
 */
export async function refreshMetricsForCompany(companyId, dateRange, io) {
  const { startDate, endDate } = getDateRange(dateRange);
  const connections = listConnections(companyId);
  const room = `company:${companyId}`;

  for (const conn of connections) {
    const full = getConnection(companyId, conn.platform);
    try {
      let data;
      if (conn.platform === 'google') {
        const { fetchGoogleMetrics } = await import('./ads.google.js');
        data = await fetchGoogleMetrics(full, startDate, endDate);
      } else if (conn.platform === 'meta') {
        const { fetchMetaMetrics } = await import('./ads.meta.js');
        data = await fetchMetaMetrics(full, startDate, endDate);
      } else if (conn.platform === 'linkedin') {
        const { fetchLinkedInMetrics } = await import('./ads.linkedin.js');
        data = await fetchLinkedInMetrics(full, startDate, endDate);
      } else {
        continue;
      }
      saveSnapshot(conn.id, dateRange, data);
      io?.to(room).emit('ads:sync:progress', { platform: conn.platform, status: 'done', data });
    } catch (err) {
      io?.to(room).emit('ads:sync:progress', { platform: conn.platform, status: 'error', error: err.message });
    }
  }

  io?.to(room).emit('ads:sync:complete', { dateRange });
}
```

import { useEffect } from 'react';
import { io as socketIo } from 'socket.io-client';
import { useAuth } from '../context/AuthContext.jsx';

// Inside AdsDashboard component:
const { user } = useAuth();

useEffect(() => {
  if (!user?.company_id) return;
  const socket = socketIo('http://localhost:3001', { withCredentials: true });
  socket.emit('join:company', user.company_id);

  socket.on('ads:sync:progress', ({ platform, status, data, error }) => {
    if (status === 'done' && data) {
      // Merge updated platform data into current state
      setData(prev => {
        if (!prev) return prev;
        const updated = prev.platforms.filter(p => p.platform !== platform);
        return { ...prev, platforms: [...updated, data] };
      });
    }
    if (status === 'error') console.warn(`[ads] ${platform} sync error:`, error);
  });

  socket.on('ads:sync:complete', () => {
    setSyncing(false);
  });

  return () => {
    socket.emit('leave:company', user.company_id);
    socket.disconnect();
  };
}, [user?.company_id]);
