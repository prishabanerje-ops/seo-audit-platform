# Exports Feature Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the `/exports` page that lets users export audit results (CSV download or Google Sheets push) for any completed crawl job, with per-severity and per-section filtering.

**Architecture:** A new Express route file (`backend/routes/exports.js`) exposes four endpoints: list jobs, list sections, stream CSV, and push to Sheets. A new service (`backend/services/sheets.service.js`) handles the Google Sheets write using `googleapis` (already installed). The React page (`frontend/src/pages/Exports.jsx`) replaces the stub and accepts a `?siteId=` query param for pre-selection from the audit view.

**Tech Stack:** Node.js ESM, Express, better-sqlite3 (sync), googleapis v171, React 18, react-router v7, lucide-react, sonner toasts, CSS custom properties.

```js
/**
 * @module routes/exports
 * Audit result exports — CSV download and Google Sheets push.
 */
import { Router } from 'express';
import db         from '../db/sqlite.js';
import { pushToSheets } from '../services/sheets.service.js';

const router = Router();

// ─── Helpers ──────────────────────────────────────────────────────────────────

function verifySite(siteId, companyId, role) {
  const site = db.prepare(
    'SELECT id, company_id, label, sheets_tab_name, sheets_spreadsheet_id FROM sites WHERE id = ?'
  ).get(siteId);
  if (!site) return { error: 'Site not found', status: 404 };
  if (role !== 'superadmin' && site.company_id !== companyId) {
    return { error: 'Access denied', status: 403 };
  }
  return { site };
}

function verifyJob(jobId, companyId, role) {
  const job = db.prepare(`
    SELECT j.id, j.site_id, j.started_at, j.completed_at, j.total_urls, j.health_score,
           s.company_id, s.label, s.sheets_tab_name, s.sheets_spreadsheet_id
    FROM crawl_jobs j
    JOIN sites s ON s.id = j.site_id
    WHERE j.id = ? AND j.status = 'completed'
  `).get(jobId);
  if (!job) return { error: 'Job not found', status: 404 };
  if (role !== 'superadmin' && job.company_id !== companyId) {
    return { error: 'Access denied', status: 403 };
  }
  return { job };
}

function buildFilterQuery(jobId, severity, sections) {
  const severityList = severity ? String(severity).split(',').filter(Boolean) : null;
  const sectionList  = sections  ? String(sections).split(',').filter(Boolean)  : null;

  let query = `
    SELECT url, section, issue_type, severity, data
    FROM audit_results
    WHERE crawl_job_id = ?
  `;
  const params = [jobId];

  if (severityList?.length) {
    query += ` AND severity IN (${severityList.map(() => '?').join(',')})`;
    params.push(...severityList);
  }
  if (sectionList?.length) {
    query += ` AND section IN (${sectionList.map(() => '?').join(',')})`;
    params.push(...sectionList);
  }

  query += ' ORDER BY section, severity, url';
  return { query, params };
}

// ─── GET /api/exports/jobs?siteId=x ──────────────────────────────────────────
router.get('/jobs', (req, res) => {
  try {
    const { siteId } = req.query;
    const { company_id, role } = req.user;

    if (!siteId) return res.status(400).json({ error: 'siteId is required' });

    const check = verifySite(siteId, company_id, role);
    if (check.error) return res.status(check.status).json({ error: check.error });

    const jobs = db.prepare(`
      SELECT id, started_at, completed_at, total_urls, health_score
      FROM crawl_jobs
      WHERE site_id = ? AND status = 'completed'
      ORDER BY started_at DESC
    `).all(siteId);

    res.json({ jobs });
  } catch (err) {
    console.error('[exports] GET /jobs error:', err);
    res.status(500).json({ error: 'Internal server error' });
  }
});

// ─── GET /api/exports/sections?jobId=x ───────────────────────────────────────
router.get('/sections', (req, res) => {
  try {
    const { jobId } = req.query;
    const { company_id, role } = req.user;

    if (!jobId) return res.status(400).json({ error: 'jobId is required' });

    const check = verifyJob(jobId, company_id, role);
    if (check.error) return res.status(check.status).json({ error: check.error });

    const sections = db.prepare(`
      SELECT DISTINCT section FROM audit_results
      WHERE crawl_job_id = ?
      ORDER BY section
    `).all(jobId).map(r => r.section);

    res.json({ sections });
  } catch (err) {
    console.error('[exports] GET /sections error:', err);
    res.status(500).json({ error: 'Internal server error' });
  }
});

// ─── GET /api/exports/csv?jobId=x&severity=a,b&sections=s1,s2 ────────────────
router.get('/csv', (req, res) => {
  try {
    const { jobId, severity, sections } = req.query;
    const { company_id, role } = req.user;

    if (!jobId) return res.status(400).json({ error: 'jobId is required' });

    const check = verifyJob(jobId, company_id, role);
    if (check.error) return res.status(check.status).json({ error: check.error });

    const { job } = check;
    const { query, params } = buildFilterQuery(jobId, severity, sections);

    // Derive filename from site label and crawl date
    const dateStr   = job.started_at ? new Date(job.started_at).toISOString().slice(0, 10) : 'export';
    const siteSlug  = (job.label || 'site').replace(/[^a-z0-9]/gi, '-').toLowerCase();
    const filename  = `audit-${siteSlug}-${dateStr}.csv`;

    res.setHeader('Content-Type', 'text/csv; charset=utf-8');
    res.setHeader('Content-Disposition', `attachment; filename="${filename}"`);

    // Write CSV header
    res.write('URL,Section,Issue Type,Severity,Details\r\n');

    // Stream rows using better-sqlite3 iterator — memory-efficient for large audits
    const stmt = db.prepare(query);
    for (const row of stmt.iterate(...params)) {
      const cols = [row.url, row.section, row.issue_type, row.severity, row.data || '']
        .map(v => `"${String(v == null ? '' : v).replace(/"/g, '""')}"`);
      res.write(cols.join(',') + '\r\n');
    }

    res.end();
  } catch (err) {
    console.error('[exports] GET /csv error:', err);
    // Headers may already be sent if streaming started
    if (!res.headersSent) {
      res.status(500).json({ error: 'Internal server error' });
    } else {
      res.end();
    }
  }
});

// ─── POST /api/exports/sheets ─────────────────────────────────────────────────
router.post('/sheets', async (req, res) => {
  try {
    const { jobId, severity, sections } = req.body;
    const { company_id, role } = req.user;

    if (!jobId) return res.status(400).json({ error: 'jobId is required' });

    const check = verifyJob(jobId, company_id, role);
    if (check.error) return res.status(check.status).json({ error: check.error });

    const { job } = check;

    if (!job.sheets_tab_name) {
      return res.status(400).json({ error: 'Configure a Sheets tab name in Site Settings first' });
    }
    if (!job.sheets_spreadsheet_id) {
      return res.status(400).json({ error: 'Configure a Sheets spreadsheet ID in Site Settings first' });
    }
    if (!process.env.GOOGLE_APPLICATION_CREDENTIALS) {
      return res.status(503).json({ error: 'Google Sheets not configured — set GOOGLE_APPLICATION_CREDENTIALS in .env' });
    }

    // Parse body arrays (may be string or array depending on how client sends)
    const severityStr = Array.isArray(severity) ? severity.join(',') : (severity || null);
    const sectionsStr = Array.isArray(sections)  ? sections.join(',')  : (sections  || null);

    const { query, params } = buildFilterQuery(jobId, severityStr, sectionsStr);
    const dbRows = db.prepare(query).all(...params);

    const rows = [
      ['URL', 'Section', 'Issue Type', 'Severity', 'Details'],
      ...dbRows.map(r => [r.url, r.section, r.issue_type, r.severity, r.data || '']),
    ];

    const { rowsWritten } = await pushToSheets({
      spreadsheetId: job.sheets_spreadsheet_id,
      tabName:        job.sheets_tab_name,
      rows,
    });

    res.json({ success: true, rowsWritten });
  } catch (err) {
    console.error('[exports] POST /sheets error:', err);
    // Surface Google API 403 as a helpful message
    const msg = err.message || '';
    if (err.code === 403 || err.status === 403 || msg.includes('403') || msg.toLowerCase().includes('forbidden')) {
      return res.status(400).json({ error: 'Sheets access denied — share the spreadsheet with your service account email' });
    }
    res.status(500).json({ error: msg || 'Failed to push to Google Sheets' });
  }
});

export default router;
```

```js
/**
 * @module services/sheets
 * Push audit result rows to a Google Sheet tab.
 * Uses a service account (GOOGLE_APPLICATION_CREDENTIALS).
 */
import { google } from 'googleapis';

/**
 * Clear a named tab in a Google Spreadsheet, then write all rows starting at A1.
 *
 * @param {object}   params
 * @param {string}   params.spreadsheetId  - Google Sheets file ID (from the sheet URL)
 * @param {string}   params.tabName        - Exact name of the sheet tab to overwrite
 * @param {string[][]} params.rows         - 2D array; first row should be the header
 * @returns {Promise<{ rowsWritten: number }>} Number of data rows written (header excluded)
 */
export async function pushToSheets({ spreadsheetId, tabName, rows }) {
  const auth = new google.auth.GoogleAuth({
    keyFile: process.env.GOOGLE_APPLICATION_CREDENTIALS,
    scopes:  ['https://www.googleapis.com/auth/spreadsheets'],
  });

  const sheets = google.sheets({ version: 'v4', auth });

  // Clear existing content in the tab
  await sheets.spreadsheets.values.clear({
    spreadsheetId,
    range: tabName,
  });

  // Write all rows starting at A1 (even if rows is empty — just leaves tab clear)
  if (rows.length > 0) {
    await sheets.spreadsheets.values.update({
      spreadsheetId,
      range:            `${tabName}!A1`,
      valueInputOption: 'RAW',
      requestBody:      { values: rows },
    });
  }

  return { rowsWritten: rows.length > 0 ? rows.length - 1 : 0 };
}
```

```jsx
/**
 * Exports — export audit results as CSV or push to Google Sheets.
 *
 * Entry points:
 *   /exports              — standalone; user picks site first
 *   /exports?siteId=xxx   — site pre-selected (linked from audit view)
 *
 * Flow: site picker → job picker → optional filters → Download CSV / Push to Sheets
 */
import { useState, useEffect, useCallback } from 'react';
import { useSearchParams }                  from 'react-router';
import { Download, FileSpreadsheet }        from 'lucide-react';
import { toast }                            from 'sonner';
import api                 from '../api/index.js';
import { useSites }        from '../context/SitesContext.jsx';
import { LoadingSkeleton } from '../components/LoadingSkeleton.jsx';

const SEVERITIES = ['critical', 'warning', 'info'];

export function Exports() {
  const [searchParams]          = useSearchParams();
  const { sites, loading: sitesLoading } = useSites();

  const [siteId,             setSiteId]             = useState(searchParams.get('siteId') || '');
  const [jobs,               setJobs]               = useState([]);
  const [jobId,              setJobId]              = useState('');
  const [sections,           setSections]           = useState([]);
  const [selectedSeverities, setSelectedSeverities] = useState(new Set(SEVERITIES));
  const [allSections,        setAllSections]        = useState(true);
  const [selectedSections,   setSelectedSections]   = useState(new Set());
  const [loadingJobs,        setLoadingJobs]        = useState(false);
  const [loadingSections,    setLoadingSections]    = useState(false);
  const [downloading,        setDownloading]        = useState(false);
  const [pushing,            setPushing]            = useState(false);

  // Load completed crawl jobs when site changes
  const loadJobs = useCallback(async (sid) => {
    setLoadingJobs(true);
    setJobs([]);
    setJobId('');
    setSections([]);
    try {
      const res = await api.get('/exports/jobs', { params: { siteId: sid } });
      setJobs(res.data.jobs || []);
    } catch {
      // toast shown by axios interceptor
    } finally {
      setLoadingJobs(false);
    }
  }, []);

  // Load distinct sections present in the selected job
  const loadSections = useCallback(async (jid) => {
    setLoadingSections(true);
    setSections([]);
    setSelectedSections(new Set());
    try {
      const res = await api.get('/exports/sections', { params: { jobId: jid } });
      const list = res.data.sections || [];
      setSections(list);
      setSelectedSections(new Set(list));
    } catch {
      // toast shown by axios interceptor
    } finally {
      setLoadingSections(false);
    }
  }, []);

  useEffect(() => { if (siteId) loadJobs(siteId); }, [siteId, loadJobs]);
  useEffect(() => { if (jobId)  loadSections(jobId); }, [jobId, loadSections]);

  // Build query params shared between CSV and Sheets
  function buildParams() {
    const sevs = [...selectedSeverities];
    const secs = allSections ? [] : [...selectedSections];
    return {
      jobId,
      ...(sevs.length < SEVERITIES.length ? { severity: sevs.join(',') } : {}),
      ...(secs.length > 0 && secs.length < sections.length ? { sections: secs.join(',') } : {}),
    };
  }

  async function downloadCsv() {
    setDownloading(true);
    try {
      const params = buildParams();
      const qs = new URLSearchParams(params).toString();
      const res = await api.get(`/exports/csv?${qs}`, { responseType: 'blob' });
      const blobUrl = URL.createObjectURL(res.data);
      const a = document.createElement('a');
      a.href = blobUrl;
      const cd = res.headers['content-disposition'] || '';
      const match = cd.match(/filename="([^"]+)"/);
      a.download = match?.[1] || 'audit-export.csv';
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
      URL.revokeObjectURL(blobUrl);
    } catch {
      toast.error('Failed to download CSV');
    } finally {
      setDownloading(false);
    }
  }

  async function pushSheets() {
    setPushing(true);
    try {
      const { jobId: jid, severity, sections: secs } = buildParams();
      const body = { jobId: jid };
      if (severity) body.severity = severity.split(',');
      if (secs)     body.sections  = secs.split(',');
      const res = await api.post('/exports/sheets', body);
      toast.success(`Pushed ${res.data.rowsWritten.toLocaleString()} rows to Sheets`);
    } catch (err) {
      toast.error(err.response?.data?.error || 'Failed to push to Sheets');
    } finally {
      setPushing(false);
    }
  }

  // Check if Sheets is configured for the selected site
  const activeSite  = (sites || []).find(s => s.id === siteId);
  const sheetsReady = !!(activeSite?.sheets_tab_name && activeSite?.sheets_spreadsheet_id);
  const canExport   = !!jobId && selectedSeverities.size > 0;

  if (sitesLoading) {
    return <div className="p-6"><LoadingSkeleton type="table" rows={4} columns={1} /></div>;
  }

  return (
    <div className="p-6 max-w-2xl mx-auto">
      <h1 className="text-xl font-bold mb-1" style={{ color: 'var(--text-primary)' }}>Exports</h1>
      <p className="text-sm mb-6" style={{ color: 'var(--text-muted)' }}>
        Download audit results as CSV or push to Google Sheets.
      </p>

      <div
        className="rounded-xl p-6 space-y-5"
        style={{ backgroundColor: 'var(--bg-surface)', border: '1px solid var(--border-default)' }}
      >
        {/* ── Step 1: Site picker ── */}
        <div>
          <label
            className="block text-xs font-semibold uppercase tracking-wide mb-1.5"
            style={{ color: 'var(--text-muted)' }}
          >
            Site
          </label>
          <select
            className="w-full rounded-lg px-3 py-2 text-sm"
            style={{
              backgroundColor: 'var(--bg-raised)',
              border: '1px solid var(--border-subtle)',
              color: 'var(--text-primary)',
            }}
            value={siteId}
            onChange={e => { setSiteId(e.target.value); setJobId(''); setSections([]); }}
          >
            <option value="">Select a site…</option>
            {(sites || []).map(s => (
              <option key={s.id} value={s.id}>{s.label}</option>
            ))}
          </select>
        </div>

        {/* ── Step 2: Crawl job picker ── */}
        <div>
          <label
            className="block text-xs font-semibold uppercase tracking-wide mb-1.5"
            style={{ color: 'var(--text-muted)' }}
          >
            Crawl Job
          </label>
          {loadingJobs ? (
            <p className="text-xs" style={{ color: 'var(--text-muted)' }}>Loading jobs…</p>
          ) : (
            <select
              className="w-full rounded-lg px-3 py-2 text-sm"
              style={{
                backgroundColor: 'var(--bg-raised)',
                border: '1px solid var(--border-subtle)',
                color: 'var(--text-primary)',
                opacity: (!siteId || jobs.length === 0) ? 0.6 : 1,
              }}
              value={jobId}
              onChange={e => setJobId(e.target.value)}
              disabled={!siteId || jobs.length === 0}
            >
              <option value="">
                {!siteId
                  ? 'Select a site first'
                  : jobs.length === 0
                    ? 'No completed crawls yet'
                    : 'Select a crawl job…'}
              </option>
              {jobs.map(j => (
                <option key={j.id} value={j.id}>
                  {new Date(j.started_at).toLocaleString(undefined, {
                    month: 'short', day: 'numeric', year: 'numeric',
                    hour: '2-digit', minute: '2-digit',
                  })}
                  {' · '}{(j.total_urls || 0).toLocaleString()} URLs
                  {j.health_score != null ? ` · Score ${j.health_score}` : ''}
                </option>
              ))}
            </select>
          )}
        </div>

        {/* ── Step 3: Filters (only shown after a job is selected) ── */}
        {jobId && (
          <>
            {/* Severity checkboxes */}
            <div>
              <p
                className="text-xs font-semibold uppercase tracking-wide mb-2"
                style={{ color: 'var(--text-muted)' }}
              >
                Severity
              </p>
              <div className="flex gap-4">
                {SEVERITIES.map(sev => (
                  <label
                    key={sev}
                    className="flex items-center gap-1.5 cursor-pointer text-sm select-none"
                    style={{ color: 'var(--text-secondary)' }}
                  >
                    <input
                      type="checkbox"
                      checked={selectedSeverities.has(sev)}
                      onChange={e => {
                        const next = new Set(selectedSeverities);
                        if (e.target.checked) next.add(sev); else next.delete(sev);
                        setSelectedSeverities(next);
                      }}
                    />
                    {sev.charAt(0).toUpperCase() + sev.slice(1)}
                  </label>
                ))}
              </div>
            </div>

            {/* Section filter */}
            <div>
              <div className="flex items-center justify-between mb-2">
                <p
                  className="text-xs font-semibold uppercase tracking-wide"
                  style={{ color: 'var(--text-muted)' }}
                >
                  Sections
                </p>
                <label
                  className="flex items-center gap-1.5 cursor-pointer text-xs select-none"
                  style={{ color: 'var(--text-secondary)' }}
                >
                  <input
                    type="checkbox"
                    checked={allSections}
                    onChange={e => setAllSections(e.target.checked)}
                  />
                  All sections
                </label>
              </div>

              {!allSections && (
                loadingSections ? (
                  <p className="text-xs" style={{ color: 'var(--text-muted)' }}>Loading sections…</p>
                ) : (
                  <div
                    className="grid grid-cols-2 gap-1 max-h-40 overflow-y-auto rounded-lg p-2"
                    style={{ backgroundColor: 'var(--bg-raised)', border: '1px solid var(--border-subtle)' }}
                  >
                    {sections.map(sec => (
                      <label
                        key={sec}
                        className="flex items-center gap-1.5 cursor-pointer text-xs select-none"
                        style={{ color: 'var(--text-secondary)' }}
                      >
                        <input
                          type="checkbox"
                          checked={selectedSections.has(sec)}
                          onChange={e => {
                            const next = new Set(selectedSections);
                            if (e.target.checked) next.add(sec); else next.delete(sec);
                            setSelectedSections(next);
                          }}
                        />
                        {sec}
                      </label>
                    ))}
                  </div>
                )
              )}
            </div>
          </>
        )}

        {/* ── Step 4: Export actions ── */}
        <div
          className="flex items-center gap-3 pt-3"
          style={{ borderTop: '1px solid var(--border-subtle)' }}
        >
          {/* CSV download */}
          <button
            type="button"
            onClick={downloadCsv}
            disabled={!canExport || downloading}
            className="inline-flex items-center gap-2 px-4 py-2 rounded-lg text-sm font-medium transition-opacity"
            style={{
              backgroundColor: 'var(--brand)',
              color: '#fff',
              opacity: (!canExport || downloading) ? 0.5 : 1,
              cursor: (!canExport || downloading) ? 'not-allowed' : 'pointer',
            }}
          >
            <Download className="w-4 h-4" />
            {downloading ? 'Downloading…' : 'Download CSV'}
          </button>

          {/* Sheets push with tooltip when not configured */}
          <div className="relative group">
            <button
              type="button"
              onClick={pushSheets}
              disabled={!canExport || !sheetsReady || pushing}
              className="inline-flex items-center gap-2 px-4 py-2 rounded-lg text-sm font-medium transition-opacity"
              style={{
                backgroundColor: 'var(--bg-raised)',
                border: '1px solid var(--border-default)',
                color: (!canExport || !sheetsReady || pushing) ? 'var(--text-muted)' : 'var(--text-primary)',
                opacity: (!canExport || !sheetsReady || pushing) ? 0.6 : 1,
                cursor: (!canExport || !sheetsReady || pushing) ? 'not-allowed' : 'pointer',
              }}
            >
              <FileSpreadsheet className="w-4 h-4" />
              {pushing ? 'Pushing…' : 'Push to Sheets'}
            </button>

            {/* Tooltip — only shown when job is selected but Sheets is not configured */}
            {jobId && !sheetsReady && (
              <div
                className="absolute bottom-full left-0 mb-2 px-2.5 py-1.5 rounded-lg text-xs whitespace-nowrap z-10 pointer-events-none opacity-0 group-hover:opacity-100 transition-opacity"
                style={{
                  backgroundColor: 'var(--bg-raised)',
                  border: '1px solid var(--border-default)',
                  color: 'var(--text-secondary)',
                }}
              >
                Configure Sheets tab name + spreadsheet ID in Site Settings first
              </div>
            )}
          </div>
        </div>
      </div>
    </div>
  );
}
```
